# Tesseract Studio (M5 / B-047): a single self-contained native C++ executable
# with a visual block canvas that drives the embedded engine directly (no
# Python). The GUI is a localhost web app served + embedded by the binary, so
# it builds and runs headless on a GPU box with no X11/OpenGL dev stack.

# --------------------------------------------------------------------------- #
# studio core: GUI-independent logic (graph model, validation, codegen,        #
# executor). Links only the engine, so it is fully unit-testable headless.     #
# --------------------------------------------------------------------------- #
add_library(tesseract_studio_core STATIC
  core/src/Json.cpp
  core/src/BlockGraph.cpp
  core/src/BlockCatalog.cpp
  core/src/Analysis.cpp
  core/src/Codegen.cpp
  core/src/Executor.cpp
)
target_include_directories(tesseract_studio_core PUBLIC
  "${CMAKE_CURRENT_SOURCE_DIR}/core/include")
target_link_libraries(tesseract_studio_core
  PUBLIC Tesseract::tesseract
  PRIVATE Tesseract::compile_options)
add_library(Tesseract::studio_core ALIAS tesseract_studio_core)

# --------------------------------------------------------------------------- #
# Embed the web UI assets (incl. the vendored LiteGraph node editor) into the  #
# binary as byte arrays (configure-time) so the executable is self-contained.  #
# --------------------------------------------------------------------------- #
set(_web_dir "${CMAKE_CURRENT_SOURCE_DIR}/app/web")

# (route ; C++ array name ; source file) for every embedded asset.
set(_asset_routes
  "/"                     "/app.js"   "/style.css"
  "/vendor/litegraph.js"  "/vendor/litegraph.css")
set(_asset_names
  kIndexHtml              kAppJs      kStyleCss
  kLiteGraphJs            kLiteGraphCss)
set(_asset_files
  "${_web_dir}/index.html"        "${_web_dir}/app.js"   "${_web_dir}/style.css"
  "${_web_dir}/vendor/litegraph.js" "${_web_dir}/vendor/litegraph.css")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${_asset_files})

set(_gen "${CMAKE_CURRENT_BINARY_DIR}/WebAssets.generated.cpp")
set(_body "// Generated by studio/CMakeLists.txt — do not edit.\n")
string(APPEND _body "#include \"tesseract/studio/WebAssets.hpp\"\n")
string(APPEND _body "namespace tesseract::studio {\n")

list(LENGTH _asset_names _n_assets)
math(EXPR _last "${_n_assets} - 1")
foreach(_i RANGE ${_last})
  list(GET _asset_names ${_i} _nm)
  list(GET _asset_files ${_i} _pa)
  file(READ "${_pa}" _hex HEX)
  string(REGEX REPLACE "(..)" "0x\\1," _arr "${_hex}")
  string(APPEND _body "static const unsigned char ${_nm}[] = {${_arr}};\n")
endforeach()

string(APPEND _body "std::string_view web_asset(std::string_view path, bool& found) {\n")
string(APPEND _body "  found = true;\n")
string(APPEND _body "  if (path == \"/\" || path == \"/index.html\")\n")
string(APPEND _body "    return {reinterpret_cast<const char*>(kIndexHtml), sizeof(kIndexHtml)};\n")
foreach(_i RANGE 1 ${_last})
  list(GET _asset_routes ${_i} _rt)
  list(GET _asset_names ${_i} _nm)
  string(APPEND _body "  if (path == \"${_rt}\")\n")
  string(APPEND _body "    return {reinterpret_cast<const char*>(${_nm}), sizeof(${_nm})};\n")
endforeach()
string(APPEND _body "  found = false; return {};\n}\n")
string(APPEND _body "std::string_view web_mime(std::string_view path) {\n")
string(APPEND _body "  if (path.size() >= 3 && path.substr(path.size()-3) == \".js\") return \"application/javascript\";\n")
string(APPEND _body "  if (path.size() >= 4 && path.substr(path.size()-4) == \".css\") return \"text/css\";\n")
string(APPEND _body "  return \"text/html; charset=utf-8\";\n}\n")
string(APPEND _body "}  // namespace tesseract::studio\n")
file(WRITE "${_gen}" "${_body}")

# --------------------------------------------------------------------------- #
# studio app: the executable (HTTP control plane + embedded UI + engine).      #
# --------------------------------------------------------------------------- #
find_package(Threads REQUIRED)
add_executable(tesseract_studio
  app/main.cpp
  app/HttpServer.cpp
  "${_gen}"
)
target_include_directories(tesseract_studio PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}/app/include")
target_link_libraries(tesseract_studio
  PRIVATE
    Tesseract::studio_core
    Tesseract::tesseract
    Tesseract::compile_options
    Threads::Threads)
set_target_properties(tesseract_studio PROPERTIES OUTPUT_NAME "tesseract_studio")

# --------------------------------------------------------------------------- #
# studio tests (headless): graph round-trip, codegen, analysis, executor.      #
